cmake_minimum_required(VERSION 3.24)
project(ExternalUniversalDataPlugin C)
find_package(UniversalData CONFIG REQUIRED COMPONENTS Contracts)
add_library(external_abs MODULE ../../plugins/abs.c)
target_include_directories(external_abs PRIVATE ../../plugins)
target_link_libraries(external_abs PRIVATE UniversalData::Contracts)
set_target_properties(external_abs PROPERTIES PREFIX "")
option(UD_BUILD_CUSTOM_TASK_PLUGIN "Build native Task using the installed shared SDK" OFF)
if(UD_BUILD_CUSTOM_TASK_PLUGIN)
  find_package(UniversalData CONFIG REQUIRED COMPONENTS Data)
  add_library(external_custom_task MODULE custom_task.c)
  target_link_libraries(external_custom_task PRIVATE UniversalData::Data)
  set_target_properties(external_custom_task PROPERTIES PREFIX "")
endif()
