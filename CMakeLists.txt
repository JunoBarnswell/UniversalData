cmake_minimum_required(VERSION 3.24)
project(UniversalData VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(UD_BUILD_TESTS "Build native behavior tests" ON)
option(UD_BUILD_HOSTS "Build CLI and isolated worker" ON)
option(UD_BUILD_JNI "Build Java JNI bridge (requires JDK)" OFF)
option(UD_BUILD_PLUGINS "Build CSV and function plugins" ON)
option(UD_WITH_DUCKDB "Build DuckDB SQL/Parquet plugin" OFF)
option(UD_WITH_SQLITE "Build SQLite connector" OFF)
option(UD_WITH_ODBC "Build ODBC connector" OFF)
option(UD_SANITIZE "Enable ASan and UBSan on GCC/Clang" OFF)
option(UD_BUILD_BENCHMARKS "Build native benchmark executables" OFF)
set(UD_COMPONENTS "Foundation;Columnar;Compute;Runtime;Contracts;Plugins;ETL;Query;Data" CACHE STRING "Components to install (dependencies are selected automatically)")
set(UD_SELECTED ${UD_COMPONENTS})
if(UD_BUILD_HOSTS OR UD_BUILD_JNI)
  list(APPEND UD_SELECTED Data)
endif()
if(UD_BUILD_PLUGINS OR UD_WITH_DUCKDB OR UD_WITH_SQLITE OR UD_WITH_ODBC)
  list(APPEND UD_SELECTED Contracts)
endif()
foreach(pass RANGE 0 8)
  foreach(component IN LISTS UD_SELECTED)
    if(component STREQUAL "Data")
      list(APPEND UD_SELECTED Foundation Columnar Compute Runtime Contracts Plugins ETL Query)
    elseif(component STREQUAL "ETL" OR component STREQUAL "Query")
      list(APPEND UD_SELECTED Compute Runtime)
    elseif(component STREQUAL "Compute" OR component STREQUAL "Runtime")
      list(APPEND UD_SELECTED Columnar)
    elseif(component STREQUAL "Plugins")
      list(APPEND UD_SELECTED Columnar Contracts)
    elseif(component STREQUAL "Columnar")
      list(APPEND UD_SELECTED Foundation)
    elseif(NOT component MATCHES "^(Foundation|Contracts)$")
      message(FATAL_ERROR "Unknown UD_COMPONENTS entry: ${component}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES UD_SELECTED)
endforeach()
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  add_compile_options(/W4 /utf-8 /experimental:c11atomics)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  add_compile_definitions(_POSIX_C_SOURCE=200809L _FILE_OFFSET_BITS=64)
  if(UD_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
function(ud_component name export)
  add_library(${name} STATIC ${ARGN})
  set_target_properties(${name} PROPERTIES EXPORT_NAME ${export})
  target_compile_definitions(${name} PUBLIC UD_STATIC)
  target_compile_features(${name} PUBLIC c_std_17)
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  add_library(UniversalData::${export} ALIAS ${name})
  install(TARGETS ${name} EXPORT UniversalDataTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${export})
endfunction()
foreach(pair "Foundation|components/foundation" "Columnar|components/data" "Compute|components/compute" "Runtime|components/execution" "Contracts|components/contracts" "Query|engines/query" "ETL|engines/etl")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 component)
  list(GET parts 1 path)
  if(component IN_LIST UD_SELECTED)
    add_subdirectory(${path})
  endif()
endforeach()
# Compile the same implementation into the facade: no static UD_STATIC export leakage.
if("Data" IN_LIST UD_SELECTED)
add_library(ud_data SHARED ${UD_ALL_SOURCES})
target_include_directories(ud_data PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(ud_data PRIVATE UD_BUILD_SHARED)
target_link_libraries(ud_data PRIVATE ${CMAKE_DL_LIBS})
if(NOT WIN32)
  target_link_libraries(ud_data PRIVATE m)
endif()
set_target_properties(ud_data PROPERTIES EXPORT_NAME Data VERSION ${PROJECT_VERSION} SOVERSION 1 C_VISIBILITY_PRESET hidden)
add_library(UniversalData::Data ALIAS ud_data)
target_compile_features(ud_data PUBLIC c_std_17)
install(TARGETS ud_data EXPORT UniversalDataTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT Data)
endif()
if(UD_BUILD_PLUGINS OR UD_WITH_DUCKDB OR UD_WITH_SQLITE OR UD_WITH_ODBC)
  add_subdirectory(plugins)
endif()
if(UD_BUILD_HOSTS)
  add_subdirectory(hosts)
endif()
if(UD_BUILD_JNI)
  add_subdirectory(bindings/java/native)
endif()
if(UD_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(UD_BUILD_BENCHMARKS)
  if(NOT TARGET ud_query)
    message(FATAL_ERROR "Benchmarks require the Query component")
  endif()
  add_executable(ud_benchmark benchmarks/columnar.c)
  target_link_libraries(ud_benchmark PRIVATE ud_query)
endif()
install(DIRECTORY include/ud DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} COMPONENT Headers)
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/UniversalData)
install(DIRECTORY licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/UniversalData/licenses)
configure_package_config_file(cmake/UniversalDataConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/UniversalDataConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UniversalData)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/UniversalDataConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT UniversalDataTargets NAMESPACE UniversalData:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UniversalData)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/UniversalDataConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/UniversalDataConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/UniversalData)
set(CPACK_PACKAGE_NAME UniversalData)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_RESOURCE_FILE_LICENSE ${PROJECT_SOURCE_DIR}/LICENSE)
set(CPACK_GENERATOR ZIP TGZ)
include(CPack)
