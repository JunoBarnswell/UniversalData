foreach(pass RANGE 0 8)
  foreach(component IN LISTS UD_SELECTED)
    if(component STREQUAL "Data")
      list(APPEND UD_SELECTED Foundation Columnar Expression Compute Runtime Contracts Plugins Package)
    elseif(component STREQUAL "Package")
      list(APPEND UD_SELECTED Expression Compute Runtime Plugins)
    elseif(component STREQUAL "Expression")
      list(APPEND UD_SELECTED Columnar Plugins)
    elseif(component STREQUAL "Compute")
      list(APPEND UD_SELECTED Columnar Expression Runtime)
    elseif(component STREQUAL "Runtime")
      list(APPEND UD_SELECTED Columnar)
    elseif(component STREQUAL "Plugins")
      list(APPEND UD_SELECTED Columnar Contracts)
    elseif(component STREQUAL "Columnar")
      list(APPEND UD_SELECTED Foundation)
    elseif(NOT component MATCHES "^(Foundation|Contracts)$")
      message(FATAL_ERROR "Unknown UD_COMPONENTS entry: ${component}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES UD_SELECTED)
endforeach()

function(ud_component name export)
  add_library(${name} STATIC ${ARGN})
  set_target_properties(${name} PROPERTIES EXPORT_NAME ${export})
  target_compile_definitions(${name} PUBLIC UD_STATIC)
  target_compile_features(${name} PUBLIC c_std_17)
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  add_library(UniversalData::${export} ALIAS ${name})
  install(TARGETS ${name} EXPORT UniversalDataTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} COMPONENT ${export})
endfunction()
foreach(pair "Foundation|components/foundation" "Columnar|components/data" "Contracts|components/contracts" "Expression|components/expression" "Runtime|components/execution" "Compute|components/compute" "Package|engines/package")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 component)
  list(GET parts 1 path)
  if(component IN_LIST UD_SELECTED)
    add_subdirectory(${path})
  endif()
endforeach()
