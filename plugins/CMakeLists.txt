function(ud_plugin_target name source)
  add_library(${name} MODULE ${source})
  target_link_libraries(${name} PRIVATE ud_contracts)
  set_target_properties(${name} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
  install(TARGETS ${name} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins)
endfunction()
if(UD_BUILD_PLUGINS)
  ud_plugin_target(ud_format_csv csv.c)
  ud_plugin_target(ud_function_abs abs.c)
endif()
if(UD_WITH_SQLITE)
  set(UD_SQLITE_AMALGAMATION "" CACHE PATH "Optional external sqlite amalgamation directory")
  if(UD_SQLITE_AMALGAMATION)
    add_library(ud_sqlite_dependency STATIC ${UD_SQLITE_AMALGAMATION}/sqlite3.c)
    target_include_directories(ud_sqlite_dependency PUBLIC ${UD_SQLITE_AMALGAMATION})
    if(NOT WIN32)
      find_package(Threads REQUIRED)
      target_link_libraries(ud_sqlite_dependency PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
    add_library(SQLite::SQLite3 ALIAS ud_sqlite_dependency)
  else()
  find_package(SQLite3 3.35 REQUIRED)
  endif()
  ud_plugin_target(ud_connector_sqlite sqlite.c)
  target_link_libraries(ud_connector_sqlite PRIVATE SQLite::SQLite3)
endif()
if(UD_WITH_ODBC)
  find_package(ODBC REQUIRED)
  ud_plugin_target(ud_connector_odbc odbc.c)
  target_link_libraries(ud_connector_odbc PRIVATE ODBC::ODBC)
endif()
if(UD_WITH_DUCKDB)
  find_path(UD_DUCKDB_INCLUDE duckdb.h REQUIRED)
  find_library(UD_DUCKDB_LIBRARY NAMES duckdb REQUIRED)
  ud_plugin_target(ud_backend_duckdb duckdb.c)
  target_include_directories(ud_backend_duckdb PRIVATE ${UD_DUCKDB_INCLUDE})
  target_link_libraries(ud_backend_duckdb PRIVATE ${UD_DUCKDB_LIBRARY})
  get_filename_component(UD_DUCKDB_LIBRARY_DIR ${UD_DUCKDB_LIBRARY} DIRECTORY)
  if(WIN32)
    set(UD_DUCKDB_RUNTIME ${UD_DUCKDB_LIBRARY_DIR}/duckdb.dll)
  else()
    set(UD_DUCKDB_RUNTIME ${UD_DUCKDB_LIBRARY_DIR}/libduckdb.so)
    set_target_properties(ud_backend_duckdb PROPERTIES INSTALL_RPATH "$ORIGIN")
  endif()
  if(EXISTS ${UD_DUCKDB_RUNTIME})
    install(FILES ${UD_DUCKDB_RUNTIME} DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins)
  endif()
endif()
