function(ud_plugin_target name)
  add_library(${name} MODULE ${ARGN})
  target_link_libraries(${name} PRIVATE ud_contracts)
  set_target_properties(${name} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
  install(TARGETS ${name} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins)
endfunction()
if(UD_WITH_SQLITE)
  set(UD_SQLITE_AMALGAMATION "" CACHE PATH "Optional external sqlite amalgamation directory")
  if(UD_SQLITE_AMALGAMATION)
    add_library(ud_sqlite_dependency STATIC ${UD_SQLITE_AMALGAMATION}/sqlite3.c)
    target_include_directories(ud_sqlite_dependency PUBLIC ${UD_SQLITE_AMALGAMATION})
    if(NOT WIN32)
      find_package(Threads REQUIRED)
      target_link_libraries(ud_sqlite_dependency PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
    add_library(SQLite::SQLite3 ALIAS ud_sqlite_dependency)
  else()
  find_package(SQLite3 3.35 REQUIRED)
  endif()
  ud_plugin_target(ud_connector_sqlite sqlite.c)
  target_link_libraries(ud_connector_sqlite PRIVATE SQLite::SQLite3)
endif()

# Separate actual vendor SDKs. MariaDB compatibility libraries never satisfy the
# Oracle MySQL client requirement, even when they export mysql_* symbols.
if(UD_WITH_MYSQL)
  find_path(UD_MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql)
  find_library(UD_MYSQL_LIBRARY NAMES mysqlclient PATH_SUFFIXES mysql)
  if(NOT UD_MYSQL_INCLUDE_DIR OR NOT UD_MYSQL_LIBRARY)
    message(FATAL_ERROR "Oracle MySQL C SDK is required for UD_WITH_MYSQL")
  endif()
  file(STRINGS "${UD_MYSQL_INCLUDE_DIR}/mysql_version.h" UD_MYSQL_VENDOR REGEX "MARIADB")
  if(UD_MYSQL_VENDOR)
    message(FATAL_ERROR "MariaDB SDK cannot be used as Oracle MySQL client")
  endif()
  ud_plugin_target(ud_connector_mysql mysql.c mysql_family.c)
  target_compile_definitions(ud_connector_mysql PRIVATE UD_MYSQL_CLIENT)
  target_include_directories(ud_connector_mysql PRIVATE "${UD_MYSQL_INCLUDE_DIR}")
  target_link_libraries(ud_connector_mysql PRIVATE "${UD_MYSQL_LIBRARY}" Threads::Threads)
endif()
if(UD_WITH_MARIADB)
  find_path(UD_MARIADB_INCLUDE_DIR mariadb_version.h PATH_SUFFIXES mariadb)
  find_library(UD_MARIADB_LIBRARY NAMES mariadb)
  if(NOT UD_MARIADB_INCLUDE_DIR OR NOT UD_MARIADB_LIBRARY)
    message(FATAL_ERROR "MariaDB Connector/C SDK is required for UD_WITH_MARIADB")
  endif()
  ud_plugin_target(ud_connector_mariadb mariadb.c mysql_family.c)
  target_compile_definitions(ud_connector_mariadb PRIVATE UD_MARIADB_CLIENT)
  target_include_directories(ud_connector_mariadb PRIVATE "${UD_MARIADB_INCLUDE_DIR}")
  target_link_libraries(ud_connector_mariadb PRIVATE "${UD_MARIADB_LIBRARY}" Threads::Threads)
endif()
