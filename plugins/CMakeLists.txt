function(ud_plugin_target name)
  add_library(${name} MODULE ${ARGN})
  target_link_libraries(${name} PRIVATE ud_contracts)
  set_target_properties(${name} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
  set(destination "${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins")
  if(WIN32 AND name MATCHES "^ud_connector_(activemq|rabbitmq|kafka|rocketmq)$")
    install(TARGETS ${name} RUNTIME_DEPENDENCY_SET ${name}_runtime
      LIBRARY DESTINATION "${destination}" COMPONENT ${name}
      RUNTIME DESTINATION "${destination}" COMPONENT ${name})
  else()
    install(TARGETS ${name} LIBRARY DESTINATION "${destination}" RUNTIME DESTINATION "${destination}")
  endif()
endfunction()

function(ud_message_runtime_install name)
  if(NOT WIN32)
    return()
  endif()
  set(runtime_dirs)
  foreach(dependency IN LISTS ARGN)
    if(TARGET "${dependency}")
      get_target_property(kind "${dependency}" TYPE)
      if(kind MATCHES "^(SHARED|STATIC|MODULE|UNKNOWN)_LIBRARY$")
        list(APPEND runtime_dirs "$<TARGET_FILE_DIR:${dependency}>" "$<TARGET_FILE_DIR:${dependency}>/../bin")
      endif()
    elseif(IS_ABSOLUTE "${dependency}")
      cmake_path(GET dependency PARENT_PATH directory)
      cmake_path(SET sibling NORMALIZE "${directory}/../bin")
      list(APPEND runtime_dirs "${directory}" "${sibling}")
    endif()
  endforeach()
  cmake_path(CONVERT "$ENV{CMAKE_PREFIX_PATH}" TO_CMAKE_PATH_LIST environment_prefixes)
  foreach(prefix IN LISTS CMAKE_PREFIX_PATH environment_prefixes)
    list(APPEND runtime_dirs "${prefix}/bin")
  endforeach()
  if(VCPKG_TARGET_TRIPLET)
    foreach(installed IN ITEMS "${VCPKG_INSTALLED_DIR}" "${_VCPKG_INSTALLED_DIR}")
      if(installed)
        set(prefix "${installed}/${VCPKG_TARGET_TRIPLET}")
        list(APPEND runtime_dirs "$<$<CONFIG:Debug>:${prefix}/debug/bin>" "${prefix}/bin")
      endif()
    endforeach()
  endif()
  list(REMOVE_DUPLICATES runtime_dirs)
  # Scan the import table recursively. Keep SDK DLLs next to the module, where
  # the Windows loader can resolve them under DLL_LOAD_DIR.
  install(RUNTIME_DEPENDENCY_SET ${name}_runtime
    DIRECTORIES ${runtime_dirs}
    PRE_EXCLUDE_REGEXES "^(api|ext)-ms-win-"
    POST_EXCLUDE_REGEXES
      "[/\\\\]([Ss][Yy][Ss][Tt][Ee][Mm]32|[Ss][Yy][Ss][Ww][Oo][Ww]64|[Ww][Ii][Nn][Ss][Xx][Ss])[/\\\\]"
    RUNTIME DESTINATION "${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins" COMPONENT ${name})
endfunction()

if(UD_WITH_ROCKETMQ)
  enable_language(CXX)
  find_package(Protobuf CONFIG REQUIRED)
  find_package(gRPC CONFIG REQUIRED)
  find_package(OpenSSL REQUIRED)
  find_package(ZLIB REQUIRED)
  find_package(Threads REQUIRED)
  find_path(UD_ROCKETMQ_PROTO_ROOT apache/rocketmq/v2/service.proto REQUIRED)
  find_path(UD_PROTOBUF_PROTO_ROOT google/protobuf/timestamp.proto REQUIRED)
  set(grpc_source_dir "${CMAKE_CURRENT_BINARY_DIR}/rocketmq-proto")
  set(grpc_proto_relative_dir "apache/rocketmq/v2")
  file(GLOB grpc_proto CONFIGURE_DEPENDS "${UD_ROCKETMQ_PROTO_ROOT}/${grpc_proto_relative_dir}/*.proto")
  set(grpc_sources)
  set(grpc_headers)
  foreach(source IN LISTS grpc_proto)
    get_filename_component(name "${source}" NAME_WE)
    list(APPEND grpc_sources "${grpc_source_dir}/${grpc_proto_relative_dir}/${name}.pb.cc")
    list(APPEND grpc_headers "${grpc_source_dir}/${grpc_proto_relative_dir}/${name}.pb.h")
  endforeach()
  set(grpc_service "${grpc_source_dir}/${grpc_proto_relative_dir}/service.grpc.pb.cc")
  set(grpc_service_header "${grpc_source_dir}/${grpc_proto_relative_dir}/service.grpc.pb.h")
  add_custom_command(OUTPUT ${grpc_sources} ${grpc_headers} "${grpc_service}" "${grpc_service_header}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${grpc_source_dir}"
    COMMAND protobuf::protoc -I "${UD_ROCKETMQ_PROTO_ROOT}" -I "${UD_PROTOBUF_PROTO_ROOT}"
      --cpp_out "${grpc_source_dir}" ${grpc_proto}
    COMMAND protobuf::protoc -I "${UD_ROCKETMQ_PROTO_ROOT}" -I "${UD_PROTOBUF_PROTO_ROOT}"
      --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> --grpc_out "${grpc_source_dir}"
      "${UD_ROCKETMQ_PROTO_ROOT}/${grpc_proto_relative_dir}/service.proto"
    DEPENDS ${grpc_proto} protobuf::protoc gRPC::grpc_cpp_plugin VERBATIM)
  ud_plugin_target(ud_connector_rocketmq rocketmq.cpp ${grpc_sources} ${grpc_headers}
    "${grpc_service}" "${grpc_service_header}")
  target_compile_features(ud_connector_rocketmq PRIVATE cxx_std_17)
  target_include_directories(ud_connector_rocketmq PRIVATE "${grpc_source_dir}")
  target_link_libraries(ud_connector_rocketmq PRIVATE gRPC::grpc++ protobuf::libprotobuf OpenSSL::Crypto ZLIB::ZLIB Threads::Threads)
  ud_message_runtime_install(ud_connector_rocketmq gRPC::grpc++ gRPC::grpc protobuf::libprotobuf
    OpenSSL::Crypto OpenSSL::SSL ZLIB::ZLIB)
endif()

# Separate actual vendor SDKs. MariaDB compatibility libraries never satisfy the
# Oracle MySQL client requirement, even when they export mysql_* symbols.
if(UD_WITH_MYSQL)
  find_path(UD_MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql)
  find_library(UD_MYSQL_LIBRARY NAMES mysqlclient PATH_SUFFIXES mysql)
  if(NOT UD_MYSQL_INCLUDE_DIR OR NOT UD_MYSQL_LIBRARY)
    message(FATAL_ERROR "Oracle MySQL C SDK is required for UD_WITH_MYSQL")
  endif()
  file(STRINGS "${UD_MYSQL_INCLUDE_DIR}/mysql_version.h" UD_MYSQL_VENDOR REGEX "MARIADB")
  if(UD_MYSQL_VENDOR)
    message(FATAL_ERROR "MariaDB SDK cannot be used as Oracle MySQL client")
  endif()
  ud_plugin_target(ud_connector_mysql mysql.c mysql_family.c binlog.c)
  target_compile_definitions(ud_connector_mysql PRIVATE UD_MYSQL_CLIENT)
  target_include_directories(ud_connector_mysql PRIVATE "${UD_MYSQL_INCLUDE_DIR}")
  target_link_libraries(ud_connector_mysql PRIVATE "${UD_MYSQL_LIBRARY}" Threads::Threads ZLIB::ZLIB)
