function(ud_plugin_target name)
  add_library(${name} MODULE ${ARGN})
  target_link_libraries(${name} PRIVATE ud_contracts)
  set_target_properties(${name} PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
  install(TARGETS ${name} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins RUNTIME DESTINATION ${CMAKE_INSTALL_LIBDIR}/UniversalData/plugins)
endfunction()
if(UD_WITH_SQLITE)
  ud_plugin_target(ud_connector_sqlite sqlite.c)
  target_link_libraries(ud_connector_sqlite PRIVATE SQLite::SQLite3)
endif()
if(UD_WITH_POSTGRESQL)
  find_package(PostgreSQL 17 REQUIRED)
  ud_plugin_target(ud_connector_postgresql postgresql.c postgresql_cdc.c postgresql_metadata.c)
  target_link_libraries(ud_connector_postgresql PRIVATE PostgreSQL::PostgreSQL)
endif()

if(UD_WITH_ODBC)
  find_package(ODBC REQUIRED)
  ud_plugin_target(ud_connector_odbc odbc.c mssql_cdc.c)
  if(EXISTS "${ODBC_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}/unixodbc.h")
    target_include_directories(ud_connector_odbc PRIVATE "${ODBC_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}")
  endif()
  target_link_libraries(ud_connector_odbc PRIVATE ODBC::ODBC Threads::Threads)
  if(UD_WITH_MSSQL_BCP)
    find_path(UD_MSSQL_INCLUDE_DIR msodbcsql.h PATH_SUFFIXES include REQUIRED)
    find_library(UD_MSSQL_LIBRARY NAMES msodbcsql-18 msodbcsql PATH_SUFFIXES lib64 REQUIRED)
    target_sources(ud_connector_odbc PRIVATE mssql_bulk.c)
    target_include_directories(ud_connector_odbc PRIVATE "${UD_MSSQL_INCLUDE_DIR}")
    if(UNIX AND NOT APPLE)
      get_filename_component(UD_MSSQL_LIBRARY_DIRECTORY "${UD_MSSQL_LIBRARY}" DIRECTORY)
      get_filename_component(UD_MSSQL_LIBRARY_FILE "${UD_MSSQL_LIBRARY}" NAME)
      target_link_directories(ud_connector_odbc PRIVATE "${UD_MSSQL_LIBRARY_DIRECTORY}")
      target_link_libraries(ud_connector_odbc PRIVATE "-l:${UD_MSSQL_LIBRARY_FILE}")
    else()
      add_library(UDMicrosoftBcp UNKNOWN IMPORTED)
      set_target_properties(UDMicrosoftBcp PROPERTIES IMPORTED_LOCATION "${UD_MSSQL_LIBRARY}")
      target_link_libraries(ud_connector_odbc PRIVATE UDMicrosoftBcp)
    endif()
    target_compile_definitions(ud_connector_odbc PRIVATE UD_WITH_MSSQL_BCP)
  endif()
endif()

if(UD_WITH_MYSQL OR UD_WITH_MARIADB)
  find_package(ZLIB REQUIRED)
endif()
# Separate actual vendor SDKs. MariaDB compatibility libraries never satisfy the
# Oracle MySQL client requirement, even when they export mysql_* symbols.
if(UD_WITH_MYSQL)
  find_path(UD_MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql)
  find_library(UD_MYSQL_LIBRARY NAMES mysqlclient PATH_SUFFIXES mysql)
  if(NOT UD_MYSQL_INCLUDE_DIR OR NOT UD_MYSQL_LIBRARY)
    message(FATAL_ERROR "Oracle MySQL C SDK is required for UD_WITH_MYSQL")
  endif()
  file(STRINGS "${UD_MYSQL_INCLUDE_DIR}/mysql_version.h" UD_MYSQL_VENDOR REGEX "MARIADB")
  if(UD_MYSQL_VENDOR)
    message(FATAL_ERROR "MariaDB SDK cannot be used as Oracle MySQL client")
  endif()
  ud_plugin_target(ud_connector_mysql mysql.c mysql_family.c binlog.c)
  target_compile_definitions(ud_connector_mysql PRIVATE UD_MYSQL_CLIENT)
  target_include_directories(ud_connector_mysql PRIVATE "${UD_MYSQL_INCLUDE_DIR}")
  target_link_libraries(ud_connector_mysql PRIVATE "${UD_MYSQL_LIBRARY}" Threads::Threads ZLIB::ZLIB)
endif()
if(UD_WITH_MARIADB)
  find_path(UD_MARIADB_INCLUDE_DIR mariadb_version.h PATH_SUFFIXES mariadb)
  find_library(UD_MARIADB_LIBRARY NAMES mariadb)
  if(NOT UD_MARIADB_INCLUDE_DIR OR NOT UD_MARIADB_LIBRARY)
    message(FATAL_ERROR "MariaDB Connector/C SDK is required for UD_WITH_MARIADB")
  endif()
  ud_plugin_target(ud_connector_mariadb mariadb.c mysql_family.c binlog.c)
  target_compile_definitions(ud_connector_mariadb PRIVATE UD_MARIADB_CLIENT)
  target_include_directories(ud_connector_mariadb PRIVATE "${UD_MARIADB_INCLUDE_DIR}")
  target_link_libraries(ud_connector_mariadb PRIVATE "${UD_MARIADB_LIBRARY}" Threads::Threads ZLIB::ZLIB)
endif()
